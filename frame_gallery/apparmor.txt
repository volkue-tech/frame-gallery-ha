#include <tunables/global>

# Frame Gallery's AppArmor profile (ARCHITECTURE.md §17.6, D-129).
#
# Enforced parent and one-way task-specific child profiles (Phase 9).
# The live check must prove both allowed behavior and denied accesses:
#     journalctl _TRANSPORT="audit" -g 'apparmor="DENIED"'
# The parent changes each fresh worker into the correct child before it
# drops credentials, sets no-new-privileges or accepts an untrusted request.
profile frame_gallery flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Process control: the parent drops each worker to user 65534, hands the
  # run's workspace to that group, and ends its workers (§11.3, D-163, D-164).
  # fsetid keeps the setgid bit when the parent sets the mode of a folder it
  # has given to the worker's group, of which it is not a member; and
  # dac_override lets it read what a worker wrote (0640, owned by 65534).
  # Docker already supplies this capability, but not dac_read_search. The
  # mandatory path rules still restrict reads/writes; no cap-add is used.
  capability setuid,
  capability setgid,
  capability chown,
  capability fsetid,
  capability dac_override,
  capability kill,
  signal (send) set=(kill,term),
  signal (receive) set=(kill,term),

  # TCP to the selected museum and the TV, DNS lookups; no raw or packet
  # sockets (§17.6).
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  deny network raw,
  deny network packet,

  # S6-Overlay's paths from the official example. Scripts also need read
  # permission for their interpreter under real enforcement.
  /init rix,
  /bin/** rix,
  /usr/bin/** rix,
  /run/{s6,s6-rc*,service}/** rix,
  /package/** rix,
  /command/** rix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # The interpreter and the app; workers start the same interpreter.
  /usr/lib/** mr,
  /opt/frame-gallery/** mr,
  /etc/** r,
  /dev/null rw,
  /dev/urandom r,
  @{PROC}/@{pid}/** r,
  @{PROC}/@{pid}/attr/apparmor/current w,
  change_profile -> **//image_worker,
  change_profile -> **//tv_worker,

  # State, the RAM-backed scratch space, and the media folder (§13.1).
  /data/{,**} rwk,
  # Atomic history/ledger backups link only their own valid primary to a
  # temporary backup name. Ordinary rw permissions do not permit link(2).
  link subset /data/state/history.json.bak.tmp-* -> /data/state/history.json,
  link subset /data/state/upload_ledger.json.bak.tmp-* -> /data/state/upload_ledger.json,
  /tmp/{,**} rwk,
  /media/ r,
  /media/** r,
  /media/frame_gallery/ w,
  /media/frame_gallery/library/ w,
  /media/frame_gallery/preview/{,**} rw,

  # No shell, executable transition, writable code, state, arbitrary tmp,
  # shared-memory file, or return transition is granted to either worker.
  # setuid/setgid are used only by our bootstrap; the subsequent drop removes
  # all capabilities and PR_SET_NO_NEW_PRIVS prevents regaining them.
  profile image_worker flags=(attach_disconnected,mediate_deleted) {
    capability setuid,
    capability setgid,
    signal (receive) set=(kill,term),
    deny network,
    /lib/** mr,
    /usr/lib/** mr,
    /opt/frame-gallery/** mr,
    /dev/null rw,
    /dev/urandom r,
    @{PROC}/@{pid}/** r,
    /media/frame_gallery/library/{,**} r,
    /tmp/frame-gallery/ r,
    /tmp/frame-gallery/run-*/ r,
    /tmp/frame-gallery/run-*/in/{,**} r,
    /tmp/frame-gallery/run-*/out/ rw,
    /tmp/frame-gallery/run-*/out/delivery-*.jpg rw,
  }

  profile tv_worker flags=(attach_disconnected,mediate_deleted) {
    capability setuid,
    capability setgid,
    signal (receive) set=(kill,term),
    network inet stream,
    deny network raw,
    deny network packet,
    /lib/** mr,
    /usr/lib/** mr,
    /opt/frame-gallery/** mr,
    /etc/ssl/{,**} r,
    /dev/null rw,
    /dev/urandom r,
    @{PROC}/@{pid}/** r,
    /tmp/frame-gallery/run-*/out/delivery-*.jpg r,
  }
}
